// CSV Parsing Logic for Spreadsheet Headers function parseCSV(csvText) { const lines = csvText.split(/\r\n|\n/); if (lines.length < 2) return []; const parsedAttendees = []; for (let i = 1; i < lines.length; i++) { const line = lines[i].trim(); if (!line) continue; // Handles standard comma-separated fields with quote escaping const row = line.match(/(".*?"|[^",\s]+)(?=\s*,|\s*$)/g) || line.split(','); const cleanRow = row.map(val => val.replace(/^"|"$/g, '').trim()); // Column Index Mapping const parentName = cleanRow[0] || 'Unknown'; const email = cleanRow[2] || ''; // Pull School from Column F (Index 5) const school = cleanRow[5] || 'N/A'; // Pull Translator Info from Column T (Index 19) and Column U (Index 20) const needsTranslator = cleanRow[19] || ''; const translatorLanguage = cleanRow[20] || ''; let translatorStatus = 'No'; if (needsTranslator.toLowerCase().includes('yes') || translatorLanguage) { translatorStatus = translatorLanguage ? `Yes (${translatorLanguage})` : 'Yes'; } const qrCode = cleanRow[23] || parentName; // Column X or fallback to Name parsedAttendees.push({ id: 'att_' + Date.now() + '_' + i, name: parentName, email: email, school: school, translator: translatorStatus, qrContent: qrCode, checkedIn: false, checkInTime: null, station: null }); } return parsedAttendees; }
